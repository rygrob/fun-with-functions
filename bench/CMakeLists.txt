cmake_minimum_required(VERSION 3.24)

project(adc_bench LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

include(FetchContent)
FetchContent_Declare(
    audio_bench
    GIT_REPOSITORY https://github.com/sketch-audio/audio_bench.git
    GIT_TAG 0adbf7a177fda9d44f606b3d74aafcfda5799940
    SYSTEM
)
FetchContent_MakeAvailable(audio_bench)

find_package(Threads REQUIRED)

# Build with fast-math to watch the split-pi method break (notes/fast-math-split-pi.md).
option(ADC_FAST_MATH "Compile with -ffast-math / /fp:fast" OFF)

add_executable(adc_bench src/main.cpp)
# Generated headers: bench/generated by default; point elsewhere to build
# another fit side by side, e.g. -DADC_GENERATED_DIR=generated-interpolation.
set(ADC_GENERATED_DIR "generated" CACHE STRING "directory of the generated *_approx.hpp headers")
target_include_directories(adc_bench PRIVATE src ${ADC_GENERATED_DIR})
target_link_libraries(adc_bench PRIVATE audio_bench Threads::Threads)
if(APPLE)
    # Accelerate's vvsinf: the SIMD sin reference for `adc_bench lutpoly`.
    target_link_libraries(adc_bench PRIVATE "-framework Accelerate")
endif()

if(MSVC)
    target_compile_options(adc_bench PRIVATE /W4 /permissive-)
else()
    target_compile_options(adc_bench PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

# tanh's square root in SIMD (src/rsqrt.cpp): standalone, no audio_bench.
add_executable(adc_rsqrt src/rsqrt.cpp)
target_include_directories(adc_rsqrt PRIVATE ${ADC_GENERATED_DIR})
if(NOT MSVC)
    target_compile_options(adc_rsqrt PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

# The benchmarks on the slides (simple/*.cpp): standalone, no audio_bench.
add_executable(adc_simple simple/main.cpp)
add_executable(adc_simple_cold simple/cold.cpp)
add_executable(adc_simple_loops simple/loops.cpp)
add_executable(adc_simple_estrin simple/estrin.cpp)  # background, not on the slides
foreach(target IN ITEMS adc_simple adc_simple_cold adc_simple_loops adc_simple_estrin)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 23)
    target_include_directories(${target} PRIVATE ${ADC_GENERATED_DIR})
    if(NOT MSVC)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()

if(ADC_FAST_MATH)
    target_compile_definitions(adc_bench PRIVATE ADC_FAST_MATH=1)
    target_compile_options(adc_bench PRIVATE $<IF:$<CXX_COMPILER_ID:MSVC>,/fp:fast,-ffast-math>)
endif()

# Regenerate the methods from the Python examples (needs numpy).
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_custom_target(regenerate
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..
        COMMENT "Regenerating generated/*_approx.hpp")
endif()
